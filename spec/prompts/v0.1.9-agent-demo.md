## Tool 层(3 个 · schema 跟 OpenAI function call 格式)

### 1. `Bash` · 跑任意 shell 命令

```python
{
  "type": "function",
  "function": {
    "name": "Bash",
    "description": "Run any shell command. Use for ls / cat / ffprobe / file 等查询. Returns stdout+stderr+exit code.",
    "parameters": {
      "type": "object",
      "properties": {
        "command": {"type": "string", "description": "shell command to run"},
        "timeout_sec": {"type": "integer", "default": 30}
      },
      "required": ["command"]
    }
  }
}
```
实现:`subprocess.run(["bash", "-c", command], timeout=timeout_sec, capture_output=True, text=True)` · 返回 `{"stdout": "...", "stderr": "...", "exit_code": N}`。

## 主循环(while · 抄 Claude Code)

```python
def agent_loop(task: str, max_iters: int = 10):
    messages = [{"role": "user", "content": task}]
    for i in range(max_iters):
        resp = client.chat.completions.create(
            model=os.getenv("SILICONFLOW_MODEL", "Pro/MiniMaxAI/MiniMax-M2.5"),
            messages=messages,
            tools=TOOLS,
            tool_choice="auto",
        )
        msg = resp.choices[0].message
        messages.append(msg.model_dump(exclude_none=True))

        if not msg.tool_calls:
            print("FINAL:", msg.content)
            return msg.content

        for tc in msg.tool_calls:
            name = tc.function.name
            args = json.loads(tc.function.arguments)
            print(f"→ tool: {name}({args})")
            result = dispatch(name, args)
            print(f"← result: {json.dumps(result, ensure_ascii=False)[:200]}")
            messages.append({
                "role": "tool",
                "tool_call_id": tc.id,
                "content": json.dumps(result, ensure_ascii=False)
            })
    print("max iters reached")
```
